VERILOG:

module Mealy_model(
output reg y,
input x,
input clk,
input reset
);

reg [1:0] state,next_state;

parameter S0 = 2'b00,
          S1 = 2'b01,
          S2 = 2'b10,
          S3 = 2'b11;

always @(posedge clk)
begin
if(reset == 0)
state <= S0;
else
state <= next_state;
end

always @(state or x)
begin
case(state)

S0:
if(x)
next_state = S1;
else
next_state = S0;

S1:
if(x)
next_state = S3;
else
next_state = S0;

S2:
if(x)
next_state = S2;
else
next_state = S0;

S3:
if(x)
next_state = S2;
else
next_state = S0;

default:
next_state = S0;

endcase
end

always @(state or x)
begin
case(state)

S0:
y = 1'b0;

S1,
S2,
S3:
y = ~x;

default:
y = 1'b0;

endcase
end
endmodule

`timescale 1ns/1ps

module Mealy_model_tb;

reg x,clk,reset;
wire y;

Mealy_model uut(
.y(y),
.x(x),
.clk(clk),
.reset(reset)
);

begin
$dumpfile("mealy.vcd");
$dumpvars(0,Mealy_model_tb);

clk = 0;
forever #5 clk = ~clk;
end

initial
begin
reset = 0;
x = 0;

#10 reset = 1;

x = 1; #10;
x = 1; #10;
x = 0; #10;
x = 1; #10;
x = 0; #10;

$finish;
end

initial
begin
$monitor("Time=%0t Reset=%b X=%b Y=%b State=%b",
$time,reset,x,y,uut.state);
end

endmodule
