VERILOG:

module Moore_Model(y,x,clk,reset);

input x,clk,reset;
output [1:0] y;

reg [1:0] state;

parameter S0 = 2'b00,
          S1 = 2'b01,
          S2 = 2'b10,
          S3 = 2'b11;

always @(posedge clk)
begin

if(reset == 0)
state <= S0;

else
case(state)

S0: if(x)
state <= S0;
else
state <= S1;

S1: if(x)
state <= S2;
else
state <= S3;

S2: if(x)
state <= S2;
else
state <= S3;

S3: if(x)
state <= S3;
else
state <= S0;

default:
state <= S0;

endcase
end
assign y = state;
endmodule



TESTBENCH:


`timescale 1ns/1ps

module Moore_Model_tb;

reg x;
reg clk;
reg reset;

wire [1:0] y;

Moore_Model uut(
.y(y),
.x(x),
.clk(clk),
.reset(reset)
);

initial
begin
$dumpfile("moore.vcd");
$dumpvars(0,Moore_Model_tb);
end

initial
begin
clk = 0;
forever #5 clk = ~clk;
end

initial
begin
reset = 0;
x = 0;

#10 reset = 1;

x = 0; #10;
x = 1; #10;
x = 0; #10;
x = 1; #10;
x = 1; #10;
x = 0; #10;
x = 1; #10;
x = 0; #10;

#20 $finish;
end

initial
begin
$monitor("Time=%0t Reset=%b X=%b State(Output)=%b",
$time,reset,x,y);
end

endmodule
